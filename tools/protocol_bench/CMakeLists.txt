cmake_minimum_required(VERSION 3.20)
project(svProtocolBench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Boost 1.75 CONFIG REQUIRED COMPONENTS json)
find_package(Protobuf CONFIG REQUIRED)
add_executable(sv-protocol-bench main.cpp codec.proto)
protobuf_generate(TARGET sv-protocol-bench LANGUAGE cpp)
target_include_directories(sv-protocol-bench PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(sv-protocol-bench PRIVATE Boost::json protobuf::libprotobuf)
