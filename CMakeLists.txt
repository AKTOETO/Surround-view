  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"

set(SV_FINGERPRINT_INPUT "")

  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${SV_FILE}" SV_FILE_HASH)
  string(APPEND SV_FINGERPRINT_INPUT "${SV_FILE}:${SV_FILE_HASH}\n")
endforeach()
