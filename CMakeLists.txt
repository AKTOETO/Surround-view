cmake_minimum_required(VERSION 3.20)
project(surround_view VERSION 0.4.0 LANGUAGES CXX)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SV_GPU "Build EGL/OpenGL ES renderer" ON)
option(SV_CLIENT "Build desktop Qt Quick prototype client" ON)
option(SV_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(SV_AURORA "Configure the offline Aurora SDK profile" OFF)
option(SV_PLATFORM_TEST "Build the native platform comparison executable" ON)
option(SV_PYTHON_TESTS "Register host Python tests" ON)
set(SV_QT_MAJOR "" CACHE STRING "Force Qt major version (5 or 6)")
if(SV_AURORA OR CMAKE_CROSSCOMPILING)
  set(SV_PYTHON_TESTS OFF CACHE BOOL "Register host Python tests" FORCE)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
if(SV_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Boost 1.75 CONFIG REQUIRED COMPONENTS json)
find_package(Threads REQUIRED)
find_package(glm CONFIG REQUIRED)
find_package(OpenCV CONFIG REQUIRED COMPONENTS core imgproc imgcodecs videoio)
if(OpenCV_VERSION VERSION_LESS 4.5)
  message(FATAL_ERROR "OpenCV >= 4.5 is required")
elseif(OpenCV_VERSION VERSION_LESS 5)
  find_package(OpenCV CONFIG REQUIRED COMPONENTS core calib3d imgproc imgcodecs videoio)
else()
  find_package(OpenCV CONFIG REQUIRED COMPONENTS core geometry calib imgproc imgcodecs objdetect videoio)
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(SV_SOURCE_REVISION "unknown" CACHE STRING "Source revision for report provenance")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.source-revision")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/.source-revision" SV_SOURCE_REVISION)
  string(STRIP "${SV_SOURCE_REVISION}" SV_SOURCE_REVISION)
elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.git")
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE SV_SOURCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
endif()
file(GLOB_RECURSE SV_FINGERPRINT_FILES CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"
  include/*.hpp src/*.cpp src/*.hpp src/*.vert src/*.frag src/*.qml src/*.qrc cmake/*.in)
list(APPEND SV_FINGERPRINT_FILES CMakeLists.txt)
list(SORT SV_FINGERPRINT_FILES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SV_FINGERPRINT_FILES})
set(SV_FINGERPRINT_INPUT "")
foreach(SV_FILE IN LISTS SV_FINGERPRINT_FILES)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${SV_FILE}" SV_FILE_HASH)
  string(APPEND SV_FINGERPRINT_INPUT "${SV_FILE}:${SV_FILE_HASH}\n")
endforeach()
string(SHA256 SV_SOURCE_FINGERPRINT "${SV_FINGERPRINT_INPUT}")
configure_file(cmake/build_info.hpp.in generated/sv/build_info.hpp @ONLY)
add_library(sv-core STATIC src/core/math.cpp src/core/config.cpp src/core/frame.cpp src/core/protocol.cpp)
target_include_directories(sv-core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(sv-core PUBLIC Boost::json glm::glm OpenSSL::Crypto)
add_library(sv-vision STATIC src/vision/camera.cpp src/vision/calibration.cpp)
target_include_directories(sv-vision SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(sv-vision PUBLIC sv-core ${OpenCV_LIBS})
add_library(sv-validation STATIC src/validation/report.cpp src/validation/cpu.cpp)
target_link_libraries(sv-validation PUBLIC sv-vision)
add_executable(sv-calibrate src/apps/calibrate.cpp)
target_link_libraries(sv-calibrate PRIVATE sv-validation)
add_executable(sv-scene src/apps/scene.cpp)
target_link_libraries(sv-scene PRIVATE sv-validation)
install(TARGETS sv-calibrate sv-scene RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
add_executable(sv-capture src/apps/capture.cpp)
target_link_libraries(sv-capture PRIVATE sv-validation)
install(TARGETS sv-capture RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(SV_PLATFORM_TEST)
  add_executable(sv-platform-test src/apps/platform_test.cpp)
  target_link_libraries(sv-platform-test PRIVATE sv-validation)
  install(TARGETS sv-platform-test RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
add_executable(sv-project src/apps/project.cpp)
target_link_libraries(sv-project PRIVATE sv-vision)
if(SV_GPU)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl)
  pkg_check_modules(GLES REQUIRED IMPORTED_TARGET glesv2)
  add_library(sv-render STATIC src/render/renderer.cpp)
  target_link_libraries(sv-render PUBLIC sv-vision PkgConfig::EGL PkgConfig::GLES)
  add_library(sv-gpu-validation STATIC src/validation/render_modes.cpp)
  target_link_libraries(sv-gpu-validation PUBLIC sv-render)
  file(READ src/render/surface.vert SV_SURFACE_VERT)
  file(READ src/render/surface.frag SV_SURFACE_FRAG)
  file(READ src/render/projection.vert SV_PROJECTION_VERT)
  file(READ src/render/projection.frag SV_PROJECTION_FRAG)
  file(READ src/render/vehicle.vert SV_VEHICLE_VERT)
  file(READ src/render/vehicle.frag SV_VEHICLE_FRAG)
  configure_file(cmake/shaders.hpp.in generated/sv/shaders.hpp @ONLY)
  add_executable(sv-bench src/apps/bench.cpp)
  target_link_libraries(sv-bench PRIVATE sv-render)
  add_executable(sv-server src/apps/server.cpp)
  target_link_libraries(sv-server PRIVATE sv-render Threads::Threads)
  if(SV_PLATFORM_TEST)
    target_link_libraries(sv-platform-test PRIVATE sv-gpu-validation)
    target_compile_definitions(sv-platform-test PRIVATE SV_HAS_GPU=1)
  endif()
  install(TARGETS sv-bench sv-server RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(SV_CLIENT)
  if(SV_QT_MAJOR)
    if(NOT SV_QT_MAJOR MATCHES "^[56]$")
      message(FATAL_ERROR "SV_QT_MAJOR must be 5 or 6")
    endif()
    set(QT_VERSION_MAJOR ${SV_QT_MAJOR})
    set(QT_FOUND TRUE)
  else()
    find_package(QT NAMES Qt5 Qt6 COMPONENTS Core Gui Qml Quick Network REQUIRED)
  endif()
  if(QT_FOUND)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Gui Qml Quick Network)
    add_executable(sv-client src/client/main.cpp src/client/bridge.cpp src/client/bridge.hpp src/client/client.qrc)
    set_target_properties(sv-client PROPERTIES AUTOMOC ON AUTORCC ON)
    target_include_directories(sv-client PRIVATE src/client)
    target_link_libraries(sv-client PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Gui Qt${QT_VERSION_MAJOR}::Qml Qt${QT_VERSION_MAJOR}::Quick Qt${QT_VERSION_MAJOR}::Network)
    install(TARGETS sv-client RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  endif()
endif()
include(CTest)
if(BUILD_TESTING OR SV_PLATFORM_TEST)
  add_executable(sv-core-tests tests/core_tests.cpp)
  target_link_libraries(sv-core-tests PRIVATE sv-core)
  install(TARGETS sv-core-tests RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  if(BUILD_TESTING)
    add_test(NAME core COMMAND sv-core-tests ${CMAKE_CURRENT_SOURCE_DIR}/configs/synthetic.json)
    if(SV_GPU)
      add_executable(sv-render-mode-tests tests/render_modes.cpp)
      target_link_libraries(sv-render-mode-tests PRIVATE sv-gpu-validation)
      add_test(NAME render_modes COMMAND sv-render-mode-tests ${CMAKE_CURRENT_SOURCE_DIR}/configs/synthetic.json)
      set_tests_properties(render_modes PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1;EGL_PLATFORM=surfaceless" TIMEOUT 60)
    endif()
    if(SV_PYTHON_TESTS)
      find_package(Python3 COMPONENTS Interpreter REQUIRED)
      add_executable(sv-vision-fixture tests/vision_fixture.cpp)
      target_link_libraries(sv-vision-fixture PRIVATE sv-vision)
      add_test(NAME vision_tools COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_vision_tools.py
        $<TARGET_FILE:sv-calibrate> $<TARGET_FILE:sv-vision-fixture> $<TARGET_FILE:sv-capture>)
      add_test(NAME independent_math COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_math.py $<TARGET_FILE:sv-project> ${CMAKE_CURRENT_SOURCE_DIR}/configs/synthetic.json)
      add_test(NAME calibration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_calibration.py)
      add_test(NAME platform_reports COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_platform_reports.py)
      add_test(NAME blender_fixture COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_blender_fixture.py)
      if(SV_GPU)
        add_test(NAME replay_ipc COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_integration.py ${CMAKE_BINARY_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/configs/synthetic.json)
        set_tests_properties(replay_ipc PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1;EGL_PLATFORM=surfaceless" TIMEOUT 90)
      endif()
    endif()
  endif()
endif()
install(TARGETS sv-project RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY configs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/surround-view/configs)
install(FILES docs/prototype/README.md docs/prototype/STATUS.md DESTINATION ${CMAKE_INSTALL_DATADIR}/surround-view/docs)
install(DIRECTORY docs/engineering/ DESTINATION ${CMAKE_INSTALL_DATADIR}/surround-view/docs/engineering)
install(DIRECTORY assets/demo/ DESTINATION ${CMAKE_INSTALL_DATADIR}/surround-view/assets/demo)
