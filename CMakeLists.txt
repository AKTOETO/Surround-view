cmake_minimum_required(VERSION 3.20)
project(surround_view VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SV_GPU "Build EGL/OpenGL ES renderer" ON)
option(SV_CLIENT "Build desktop Qt Quick prototype client" ON)
option(SV_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
if(SV_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Boost CONFIG REQUIRED COMPONENTS json)
find_package(Threads REQUIRED)
find_package(glm CONFIG QUIET)
if(NOT glm_FOUND)
  include(FetchContent)
  FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.3 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(glm)
endif()
add_library(sv-core src/core/math.cpp src/core/config.cpp src/core/frame.cpp src/core/protocol.cpp)
target_include_directories(sv-core PUBLIC include)
target_link_libraries(sv-core PUBLIC Boost::json glm::glm)
add_executable(sv-project src/apps/project.cpp)
target_link_libraries(sv-project PRIVATE sv-core)
if(SV_GPU)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl)
  pkg_check_modules(GLES REQUIRED IMPORTED_TARGET glesv2)
  add_library(sv-render src/render/renderer.cpp)
  target_link_libraries(sv-render PUBLIC sv-core PkgConfig::EGL PkgConfig::GLES)
  target_compile_definitions(sv-render PRIVATE SV_SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/src/render")
  add_executable(sv-bench src/apps/bench.cpp)
  target_link_libraries(sv-bench PRIVATE sv-render)
  add_executable(sv-server src/apps/server.cpp)
  target_link_libraries(sv-server PRIVATE sv-render Threads::Threads)
endif()
if(SV_CLIENT)
  find_package(QT NAMES Qt5 Qt6 COMPONENTS Core Gui Qml Quick Network QUIET)
  if(QT_FOUND)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Gui Qml Quick Network)
    set(CMAKE_AUTOMOC ON)
    add_executable(sv-client src/client/main.cpp src/client/bridge.cpp src/client/bridge.hpp)
    target_include_directories(sv-client PRIVATE src/client)
    target_link_libraries(sv-client PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Gui Qt${QT_VERSION_MAJOR}::Qml Qt${QT_VERSION_MAJOR}::Quick Qt${QT_VERSION_MAJOR}::Network)
    target_compile_definitions(sv-client PRIVATE SV_QML_PATH="${CMAKE_CURRENT_SOURCE_DIR}/src/client/Main.qml")
  else()
    message(STATUS "Qt Quick unavailable; desktop client disabled")
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(sv-core-tests tests/core_tests.cpp)
  target_link_libraries(sv-core-tests PRIVATE sv-core)
  add_test(NAME core COMMAND sv-core-tests ${CMAKE_CURRENT_SOURCE_DIR}/configs/synthetic.json)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME independent_math COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_math.py $<TARGET_FILE:sv-project> ${CMAKE_CURRENT_SOURCE_DIR}/configs/synthetic.json)
  add_test(NAME calibration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_calibration.py)
  if(SV_GPU)
    add_test(NAME replay_ipc COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_integration.py ${CMAKE_BINARY_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/configs/synthetic.json)
    set_tests_properties(replay_ipc PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1;EGL_PLATFORM=surfaceless" TIMEOUT 90)
  endif()
endif()
